47macro_rules! read_cpu_info_keys {
48    ($key:ident, $value:ident, $cpu_info:ident {
49        $(
50            $key_name:literal => $field:ident : $type_:ty
51        ),* $(,)?
52    }) => {
53        match $key {
54            $(
55                $key_name => {
56                    let parsed: $type_ = $value.parse()?;
57
58                    if $cpu_info.$field.is_some() {
59                        return Err(Error::Generic(DUPLICATE_CPU_INFO_KEY_ERROR));
60                    }
61
62                    $cpu_info.$field = Some(parsed);
63                }
64            )*
65            _ => {}
66        }
67    };
68}

70fn parse_cpu_info(input: &str) -> Result<Vec<CpuInfo>, Error> {
71    let mut cpu_infos = Vec::new();
72    let mut current = Some(CpuInfo::new());
73    for line in input.split('\n') {
74        if line.is_empty() {
75            if let Some(info) = current.take() {
76                cpu_infos.push(info);
77            }
78            continue;
79        }
80
81        let (key, value) = line
82            .split_once(":")
83            .ok_or(Error::Generic("Line missing \":\""))?;
84        let key = key.trim();
85        let value = value.trim();
86        let current = current.get_or_insert(CpuInfo::new());
87
88        read_cpu_info_keys!(key, value, current {
89            "processor" => processor: u32,
90            "physical id" => physical_id: u32,
91        });
92    }
93
94    Ok(cpu_infos)
95}

97fn get_cpu_info() -> Result<Vec<CpuInfo>, Error> {
98    let file = std::fs::read_to_string("/proc/cpuinfo")?;
99    parse_cpu_info(&file)

109fn get_cpu_freq_info() -> Result<CpuFreqInfo, Error> {
110    // Inspired by: https://unix.stackexchange.com/questions/87522/why-do-cpuinfo-cur-freq-and-proc-cpuinfo-report-different-numbers
111
112    let mut frequencies = Vec::with_capacity(4);
113    let mut min_frequencies = Vec::with_capacity(4);
114    let mut max_frequencies = Vec::with_capacity(4);
115    for entry in std::fs::read_dir("/sys/devices/system/cpu/")? {
116        let entry = entry?;
117        let file_name = entry
118            .file_name()
119            .to_os_string()
120            .into_string()
121            .map_err(|_| Error::Generic("File name is not utf8"))?;
122        let trimmed_file_name = match file_name.strip_prefix("cpu") {
123            Some(value) => value,
124            None => continue,
125        };
126        let _cpu_n = match trimmed_file_name.parse::<u32>() {
127            Ok(value) => value,
128            Err(_error) => continue,
129        };
130
131        let entry_path = entry.path();
132        let cpufreq_path = entry_path.join("cpufreq");
133        // Github Actions seems to not provide this folder.
134        if !cpufreq_path.try_exists()? {
135            continue;
136        }
137
138        let cpuinfo_cur_freq_path = cpufreq_path.join("cpuinfo_cur_freq");
139        let scaling_cur_freq_path = cpufreq_path.join("scaling_cur_freq");
140        let cpu_freq_string = std::fs::read_to_string(cpuinfo_cur_freq_path)
141            .or_else(|_| std::fs::read_to_string(scaling_cur_freq_path))?;
142        let cpu_freq: u64 = cpu_freq_string.trim().parse()?;
143        frequencies.push(cpu_freq);
144
145        let min_freq_path = cpufreq_path.join("cpuinfo_min_freq");
146        let min_freq_string = std::fs::read_to_string(min_freq_path)?;
147        let min_cpu_freq: u64 = min_freq_string.trim().parse()?;
148        min_frequencies.push(min_cpu_freq);
149
150        let max_freq_path = cpufreq_path.join("cpuinfo_max_freq");
151        let max_freq_string = std::fs::read_to_string(max_freq_path)?;
152        let max_cpu_freq: u64 = max_freq_string.trim().parse()?;
153        max_frequencies.push(max_cpu_freq);
154    }
155    if frequencies.is_empty() {
156        return Err(Error::MissingValue);
157    }
158    let cpu_count =
159        f64::from(u32::try_from(frequencies.len()).expect("CPU count cannot fit in a u32"));
160
161    let curr_sum: f64 = frequencies.iter().map(|value| *value as f64).sum();
162    let curr_freq_khz = curr_sum / cpu_count;
163    let curr_freq_hz = curr_freq_khz * 1_000.0;
164
165    let min_sum: f64 = min_frequencies.iter().map(|value| *value as f64).sum();
166    let min_freq_khz = min_sum / cpu_count;
167    let min_freq_hz = min_freq_khz * 1_000.0;
168
169    let max_sum: f64 = max_frequencies.iter().map(|value| *value as f64).sum();
170    let max_freq_khz = max_sum / cpu_count;
171    let max_freq_hz = max_freq_khz * 1_000.0;
172
173    Ok(CpuFreqInfo {
174        freq: curr_freq_hz as u64,
175        min_freq: min_freq_hz as u64,
176        max_freq: max_freq_hz as u64,
177    })
178}

180#[derive(Debug)]
181struct Stat {
182    user: u64,
183    nice: u64,
184    system: u64,
185    idle: u64,
186    iowait: u64,
187    irq: u64,
188    softirq: u64,
189    steal: u64,
190    #[expect(dead_code)]
191    guest: u64,
192    #[expect(dead_code)]
193    guest_nice: u64,
194}
195
196impl Stat {
197    /// Get the total time.
198    fn total(&self) -> u64 {
199        self.user
200            + self.nice
201            + self.system
202            + self.idle
203            + self.iowait
204            + self.irq
205            + self.softirq
206            + self.steal
207    }
208
209    /// Get the idle time.
210    fn idle(&self) -> u64 {
211        self.idle + self.iowait
212    }
213}

215fn get_stat() -> Result<Stat, Error> {
216    const MISSING_PART_ERROR: &str = "Missing stat part";
217
218    let stat_string = std::fs::read_to_string("/proc/stat")?;
219    let (line, _) = stat_string
220        .split_once('\n')
221        .ok_or(Error::Generic("Missing newline"))?;
222
223    let mut iter = line.split(' ').filter(|value| !value.is_empty());
224    let cpu = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
225    if cpu != "cpu" {
226        return Err(Error::Generic("Invalid cpu part"));
227    }
228
229    let user = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
230    let user: u64 = user.parse()?;
231
232    let nice = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
233    let nice: u64 = nice.parse()?;
234
235    let system = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
236    let system: u64 = system.parse()?;
237
238    let idle = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
239    let idle: u64 = idle.parse()?;
240
241    let iowait = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
242    let iowait: u64 = iowait.parse()?;
243
244    let irq = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
245    let irq: u64 = irq.parse()?;
246
247    let softirq = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
248    let softirq: u64 = softirq.parse()?;
249
250    let steal = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
251    let steal: u64 = steal.parse()?;
252
253    let guest = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
254    let guest: u64 = guest.parse()?;
255
256    let guest_nice = iter.next().ok_or(Error::Generic(MISSING_PART_ERROR))?;
257    let guest_nice: u64 = guest_nice.parse()?;
258
259    if iter.next().is_some() {
260        return Err(Error::Generic("Unexpected stat part"));
261    }
262
263    Ok(Stat {
264        user,
265        nice,
266        system,
267        idle,
268        iowait,
269        irq,
270        softirq,
271        steal,
272        guest,
273        guest_nice,
274    })
275}

277fn calc_cpu_usage(start: &Stat, end: &Stat) -> Result<f64, Error> {
278    let start_total = start.total();
279    let start_idle = start.idle();
280
281    let end_total = end.total();
282    let end_idle = end.idle();
283
284    let total = (end_total - start_total) as f64;
285    let idle = (end_idle - start_idle) as f64;
286
287    if total == 0.0 {
288        return Err(Error::Generic("total was 0"));
289    }
290
291    Ok(100.0 * (1.0 - (idle / total)))
292}

300fn get_thermal_zones() -> Result<Vec<ThermalZone>, Error> {
301    const ENODATA: i32 = 61;
302
303    let mut zones = Vec::with_capacity(4);
304    for entry in std::fs::read_dir("/sys/class/thermal")? {
305        let entry = entry?;
306        let file_name = entry
307            .file_name()
308            .into_string()
309            .map_err(|_| Error::Generic("File name is not utf8"))?;
310        let file_name = match file_name.strip_prefix("thermal_zone") {
311            Some(name) => name,
312            None => continue,
313        };
314        let _n: u64 = file_name.parse()?;
315
316        let entry_path = entry.path();
317        let temp_path = entry_path.join("temp");
318        let temp_string = match std::fs::read_to_string(temp_path) {
319            Ok(value) => value,
320            Err(error) if error.raw_os_error() == Some(ENODATA) => continue,
321            Err(error) => return Err(Error::Io(error)),
322        };
323        let temp: u64 = temp_string.trim().parse()?;
324
325        let type_path = entry_path.join("type");
326        let mut type_string = std::fs::read_to_string(type_path)?;
327        if type_string.ends_with('\n') {
328            type_string.pop();
329        }
330
331        zones.push(ThermalZone {
332            name: type_string,
333            value: temp,
334        });
335    }
336
337    Ok(zones)
338}

348pub(crate) struct CacheContext {
349    sysinfo: OnceCell<SysInfo>,
350    utsname: OnceCell<UtsName>,
351    cpu_info: OnceCell<Vec<CpuInfo>>,

362            utsname: OnceCell::new(),
363            cpu_info: OnceCell::new(),
364            cpu_freq_info: OnceCell::new(),
365            stat_info: OnceCell::new(),

374    }

408    /// Start measuring.
409    pub fn start(&self) -> Result<(), Error> {
410        self.stat_info.get_or_try_init(|| {
411            let start = get_stat()?;
412
413            Ok::<_, Error>(StatInfo {
414                start,
415                end: OnceCell::new(),

417        })?;
418        Ok(())
419    }

421    /// End measuring.
422    pub fn end(&self) -> Result<(), Error> {
423        let stat_info = self.get_stat_info()?;
424        stat_info.end.get_or_try_init(get_stat)?;
425        Ok(())
426    }

444    /// Get the architecture.
445    pub fn get_architecture(&self) -> Result<Option<Arch>, Error> {
446        let utsname = self.get_utsname()?;
447
448        // See:
449        // * https://en.wikipedia.org/wiki/Uname
450        // * https://stackoverflow.com/questions/45125516/possible-values-for-uname-m
451        match utsname.machine().to_str().ok_or(Error::InvalidUtf8OsStr)? {
452            "x86" | "i386" | "i586" | "i686" => Ok(Some(Arch::X86)),
453            "x86_64" | "amd64" => Ok(Some(Arch::X86_64)),
454            "arm" | "armv6l" | "armv7" | "armv7l" => Ok(Some(Arch::Arm)),
455            "aarch64_be" | "aarch64" | "armv8b" | "armv8l" => Ok(Some(Arch::AArch64)),
456            "ppc" => Ok(Some(Arch::PowerPc)),
457            "ppc64" | "ppc64le" => Ok(Some(Arch::PowerPc64)),
458            "sparc64" => Ok(Some(Arch::Sparc64)),
459            _ => Ok(None),
460        }
461    }

513    /// Get the number of physical cpu cores.
514    pub fn count_physical_cpus(&self) -> Result<usize, Error> {
515        let cpu_info = self.get_cpu_info()?;
516
517        // Some computers, like the Raspberry Pi 4, will not report the physical_id field.
518        // In these cases, fallback to counting based on the processor field.
519        let has_physical_ids = cpu_info.iter().any(|info| info.physical_id.is_some());
520        if has_physical_ids {
521            let physical_ids: HashSet<u32> = cpu_info
522                .iter()
523                .map(|info| {
524                    info.physical_id
525                        .ok_or(Error::Generic("Missing physical id"))
526                })
527                .collect::<Result<_, _>>()?;
528            Ok(physical_ids.len())
529        } else {
530            let processors: HashSet<u32> = cpu_info
531                .iter()
532                // The Raspberry Pi 4 will report the BCM2835 here (why?),
533                // which does not have a processor field.
534                .filter_map(|info| info.processor)
535                .collect();
536            let count = processors.len();
537            if count == 0 {
538                return Err(Error::Generic("Count should be greater than 0"));
539            }
540
541            Ok(count)
542        }
543    }
