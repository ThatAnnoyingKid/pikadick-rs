102fn rusqlite_log_handler(error_code: i32, message: &str) {
103    use nd_async_rusqlite::rusqlite::ffi::{
104        Error,
105        SQLITE_NOTICE,
106        SQLITE_WARNING,
107    };
108
109    let error = Error::new(error_code);
110
111    match error_code & 0xff {
112        // Couldn't figure out how to dedupe this..
113        SQLITE_NOTICE => tracing::info!(
114            target: "sqlite",
115            code = %error,
116            "{message}",
117        ),
118        SQLITE_WARNING => tracing::warn!(
119            target: "sqlite",
120            code = %error,
121            "{message}",
122        ),
123        _ => tracing::error!(
124            target: "sqlite",
125            code = %error,
126            "{message}",
127        ),
128    }
129}

131async fn handle_ctrl_c(shard_manager: Arc<ShardManager>) {
132    match tokio::signal::ctrl_c()
133        .await
134        .context("Failed to set ctrl-c handler")
135    {
136        Ok(_) => {
137            info!("Shutting down...");
138            info!("Stopping client...");
139            shard_manager.shutdown_all().await;
140        }
141        Err(error) => {
142            error!("{error}");
143            shard_manager.shutdown_all().await;
144        }
145    };
146}

148async fn process_urls_in_message(
149    ctx: &FrameworkContext<'_, PoiseData, PoiseError>,
150    msg: &Message,
151) -> anyhow::Result<()> {
152    #[derive(Debug, Copy, Clone, Eq, PartialEq, Hash)]
153    enum UrlType {
154        TikTok,
155    }
156
157    let tiktok_data = &ctx.user_data.tiktok_data;
158
159    // Only embed guild links
160    let guild_id = match msg.guild_id {
161        Some(id) => id,
162        None => {
163            return Ok(());
164        }
165    };
166
167    // No Bots
168    if msg.author.bot {
169        return Ok(());
170    }
171
172    let mut embed_url_types = HashSet::new();
173    let tiktok_embed_flags = ctx
174        .user_data
175        .database
176        .get_tiktok_embed_flags(guild_id)
177        .await
178        .with_context(|| format!("Failed to get tiktok-embed server data for {guild_id}"))
179        .unwrap_or_else(|error| {
180            error!("{error:?}");
181            TikTokEmbedFlags::empty()
182        });
183    if tiktok_embed_flags.contains(TikTokEmbedFlags::ENABLED) {
184        embed_url_types.insert(UrlType::TikTok);
185    }
186
187    // If there are no url types to embed, exit.
188    if embed_url_types.is_empty() {
189        return Ok(());
190    }
191
192    // Extract urls.
193    let urls: Vec<(UrlType, Url)> = crate::util::extract_urls(&msg.content)
194        .filter_map(|url| {
195            let url_host = match url.host() {
196                Some(host) => host,
197                None => return None,
198            };
199
200            let url_type = match url_host {
201                url::Host::Domain("vm.tiktok.com" | "tiktok.com" | "www.tiktok.com") => {
202                    UrlType::TikTok
203                }
204                _ => return None,
205            };
206
207            Some((url_type, url))
208        })
209        .filter(|(url_type, _url)| embed_url_types.contains(url_type))
210        .collect();
211
212    if urls.is_empty() {
213        return Ok(());
214    }
215
216    let mut loading_reaction = Some(LoadingReaction::new(ctx.serenity_context.http.clone(), msg));
217    for (url_type, url) in urls.iter() {
218        match url_type {
219            UrlType::TikTok => {
220                tiktok_data
221                    .try_embed_url(
222                        ctx.serenity_context,
223                        msg,
224                        url,
225                        &mut loading_reaction,
226                        tiktok_embed_flags.contains(TikTokEmbedFlags::DELETE_LINK),
227                    )
228                    .await
229                    .context("Failed to generate tiktok embed")?;
230            }
231        }
232    }
233
234    Ok(())
235}

237async fn set_status(ctx: &FrameworkContext<'_, PoiseData, PoiseError>) -> anyhow::Result<()> {
238    let config = &ctx.user_data.config;
239    let ctx = ctx.serenity_context;
240
241    if let (Some(status), Some(kind)) = (config.status_name(), config.status_type()) {
242        match kind {
243            ActivityKind::Listening => {
244                ctx.set_activity(Some(ActivityData::listening(status)));
245            }
246            ActivityKind::Streaming => {
247                let result = async {
248                    let activity = ActivityData::streaming(
249                        status,
250                        config.status_url().context("Failed to get status url")?,
251                    )?;
252
253                    ctx.set_activity(Some(activity));
254
255                    anyhow::Ok(())
256                }
257                .await;
258
259                if let Err(error) = result.context("Failed to set activity") {
260                    error!("{error:?}");
261                }
262            }
263            ActivityKind::Playing => {
264                ctx.set_activity(Some(ActivityData::playing(status)));
265            }
266        }
267    }
268
269    Ok(())
270}

272/// Set up a serenity client
273async fn setup_client(poise_data: PoiseData) -> anyhow::Result<Client> {
274    let config = poise_data.config.clone();
275
276    let framework_options = poise::FrameworkOptions {
277        commands: vec![
278            self::commands::deviantart(),
279            self::commands::iqdb(),
280            self::commands::latency(),
281            self::commands::leave(),
282            self::commands::nekos(),
283            self::commands::ping(),
284            self::commands::quizizz(),
285            self::commands::r6tracker(),
286            self::commands::rule34(),
287            self::commands::sauce_nao(),
288            self::commands::stop(),
289            self::commands::system(),
290            self::commands::tic_tac_toe(),
291            self::commands::tiktok_embed(),
292            self::commands::urban(),
293            self::commands::uwuify(),
294            self::commands::vaporwave(),
295            self::commands::xkcd(),
296            self::commands::yodaspeak(),
297            self::commands::zalgo(),
298        ],
299        on_error: |error| {
300            (async move {
301                match error {
302                    FrameworkError::CommandCheckFailed { ctx, .. } => {
303                        let result = ctx.reply("Command check failed").await;
304                        if let Err(error) = result {
305                            error!("{error}");
306                        }
307                    }
308                    FrameworkError::NsfwOnly { ctx, .. } => {
309                        let result = ctx
310                            .reply("This command can only be used in nsfw channels.")
311                            .await;
312                        if let Err(error) = result {
313                            error!("{error}");
314                        }
315                    }
316                    FrameworkError::Command { ctx, error, .. } => {
317                        warn!("{error:?}");
318
319                        let result = ctx.reply(format!("{error}")).await;
320                        if let Err(error) = result {
321                            error!("{error}");
322                        }
323                    }
324                    FrameworkError::CooldownHit {
325                        ctx,
326                        remaining_cooldown,
327                        ..
328                    } => {
329                        let seconds = remaining_cooldown.as_secs_f64();
330                        let result = ctx
331                            .reply(format!(
332                                "Wait {seconds:2} seconds to use that command again."
333                            ))
334                            .await;
335                        if let Err(error) = result {
336                            error!("{error}");
337                        }
338                    }
339                    _ => {
340                        error!("{error}");
341
342                        if let Some(ctx) = error.ctx() {
343                            let result = ctx.reply(format!("{error}")).await;
344                            if let Err(error) = result {
345                                error!("{error}");
346                            }
347                        }
348                    }
349                }
350            })
351            .boxed()
352        },
353        event_handler: |context, event| {
354            async move {
355                match event {
356                    FullEvent::Message { new_message } => {
357                        process_urls_in_message(&context, new_message).await?;
358                    }
359                    FullEvent::Ready { data_about_bot } => {
360                        set_status(&context).await?;
361                        info!("Logged in as \"{}\"", data_about_bot.user.name);
362                    }
363                    FullEvent::Resume { .. } => {
364                        warn!("Resumed connection");
365                    }
366                    _ => {}
367                }
368                anyhow::Ok(())
369            }
370            .boxed()
371        },
372        ..Default::default()
373    };
374
375    let framework = poise::Framework::builder()
376        .options(framework_options)
377        .setup(move |ctx, _ready, framework| {
378            async move {
379                poise::builtins::register_globally(ctx, &framework.options().commands).await?;
380                if let Some(test_guild_id) = poise_data.config.test_guild {
381                    poise::builtins::register_in_guild(
382                        ctx,
383                        &framework.options().commands,
384                        test_guild_id,
385                    )
386                    .await?;
387                }
388                info!("Registered commands");
389
390                anyhow::Ok(poise_data)
391            }
392            .boxed()
393        })
394        .build();
395
396    // Build the client
397    let client = Client::builder(
398        config.token.clone(),
399        GatewayIntents::non_privileged() | GatewayIntents::MESSAGE_CONTENT,
400    )
401    .application_id(ApplicationId::new(config.application_id))
402    .framework(framework)
403    .register_songbird()
404    .await
405    .context("Failed to create client")?;
406
407    // TODO: Spawn a task for this earlier?
408    // Spawn the ctrl-c handler
409    tokio::spawn(handle_ctrl_c(client.shard_manager.clone()));
410
411    Ok(client)
412}

414/// Data from the setup function
415struct SetupData {
416    tokio_rt: tokio::runtime::Runtime,
417    config: Arc<Config>,
418    lock_file: AsyncLockFile,
419    worker_guard: WorkerGuard,
420}
421
422/// Pre-main setup
423fn setup(cli_options: CliOptions) -> anyhow::Result<SetupData> {
424    eprintln!("Starting tokio runtime...");
425    let tokio_rt = tokio::runtime::Builder::new_multi_thread()
426        .enable_all()
427        .build()
428        .context("Failed to start tokio runtime")?;
429
430    let config = setup::load_config(&cli_options.config)
431        .map(Arc::new)
432        .context("Failed to load config")?;
433
434    eprintln!("Opening data directory...");
435    let data_dir_metadata = match std::fs::metadata(&config.data_dir) {
436        Ok(metadata) => Some(metadata),
437        Err(e) if e.kind() == std::io::ErrorKind::NotFound => None,
438        Err(e) => {
439            return Err(e).context("Failed to get metadata for the data dir");
440        }
441    };
442
443    let _missing_data_dir = data_dir_metadata.is_none();
444    match data_dir_metadata.as_ref() {
445        Some(metadata) => {
446            if metadata.is_dir() {
447                eprintln!("Data directory already exists.");
448            } else if metadata.is_file() {
449                bail!("Failed to create or open data directory, the path is a file");
450            }
451        }
452        None => {
453            eprintln!("Data directory does not exist. creating...");
454            std::fs::create_dir_all(&config.data_dir).context("Failed to create data directory")?;
455        }
456    }
457
458    eprintln!("Creating lockfile...");
459    let lock_file_path = config.data_dir.join("pikadick.lock");
460    let lock_file = AsyncLockFile::blocking_open(lock_file_path.as_std_path())
461        .context("Failed to open lockfile")?;
462    let lock_file_locked = lock_file
463        .try_lock_with_pid_blocking()
464        .context("Failed to try to lock the lockfile")?;
465    ensure!(lock_file_locked, "Another process has locked the lockfile");
466
467    std::fs::create_dir_all(config.log_file_dir()).context("Failed to create log file dir")?;
468    std::fs::create_dir_all(config.cache_dir()).context("Failed to create cache dir")?;
469
470    // Everything past here is assumed to need tokio
471    let _enter_guard = tokio_rt.handle().enter();
472
473    eprintln!("Setting up logger...");
474    let worker_guard = logger::setup(&config).context("Failed to initialize logger")?;
475
476    eprintln!();
477    Ok(SetupData {
478        tokio_rt,
479        config,
480        lock_file,
481        worker_guard,
482    })
483}

485/// The async entry
486async fn async_main(config: Arc<Config>) -> anyhow::Result<()> {
487    const DATABASE_NAME: &str = "pikadick.sqlite";
488
489    let database_path = config.data_dir.join(DATABASE_NAME);
490    let database = Database::new(database_path)
491        .await
492        .context("Failed to open database")?;
493
494    // TODO: See if it is possible to start serenity without a network
495    info!("Setting up client...");
496    let poise_data = Arc::new(PoiseDataInner::new(config.clone(), database.clone()).await?);
497    let mut client = setup_client(poise_data.clone())
498        .await
499        .context("Failed to set up client")?;
500
501    info!("Logging in...");
502    client.start().await.context("Failed to run client")?;
503    drop(client);
504
505    info!("Running shutdown routine for client data...");
506    poise_data.shutdown().await;
507    drop(poise_data);
508
509    info!("Closing database...");
510    database.close().await.context("Failed to close database")?;
511
512    Ok(())
513}

515/// The actual entry point
516fn real_main(setup_data: SetupData) -> anyhow::Result<()> {
517    // We spawn this is a seperate thread/task as the main thread does not have enough stack space
518    let _enter_guard = setup_data.tokio_rt.enter();
519    let result = setup_data
520        .tokio_rt
521        .block_on(tokio::spawn(async_main(setup_data.config)));
522
523    let shutdown_start = Instant::now();
524    info!("Shutting down tokio runtime (shutdown timeout is {TOKIO_RT_SHUTDOWN_TIMEOUT:?})...");
525    setup_data
526        .tokio_rt
527        .shutdown_timeout(TOKIO_RT_SHUTDOWN_TIMEOUT);
528    let elapsed = shutdown_start.elapsed();
529    info!("Shutdown tokio runtime in {elapsed:?}");
530
531    info!("Unlocking lockfile...");
532    setup_data
533        .lock_file
534        .blocking_unlock()
535        .context("Failed to unlock lockfile")?;
536
537    info!("Successful shutdown");
538
539    // Logging no longer reliable past this point
540    drop(setup_data.worker_guard);
541
542    result?
543}

545/// The main entry.
546///
547/// Sets up the program and calls `real_main`.
548/// This allows more things to drop correctly.
549/// This also calls setup operations like loading config and setting up the tokio runtime,
550/// logging errors to the stderr instead of the loggers, which are not initialized yet.
551fn main() -> anyhow::Result<()> {
552    // This line MUST run first.
553    // It is needed to exit early if the options are invalid,
554    // and this will NOT run destructors if it does so.
555    let cli_options = argh::from_env();
556
557    // Safety:
558    // 1. SQLite has not been called yet.
559    // 2. The logging callback does not invoke SQLite.
560    // 3. The logging callback is threadsafe.
561    unsafe {
562        nd_async_rusqlite::rusqlite::trace::config_log(Some(rusqlite_log_handler))
563            .context("Failed to install sqlite log handler")?;
564    }
565
566    let setup_data = setup(cli_options)?;
567    real_main(setup_data)?;
568
569    Ok(())
