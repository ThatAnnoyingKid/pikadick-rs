25fn format_boot_time(value: SystemTime) -> anyhow::Result<String> {
26    let timestamp = jiff::Timestamp::try_from(value).context("Failed to make timestamp")?;
27
28    Ok(timestamp.strftime("%e %b %Y, %I:%M:%S %p UTC").to_string())
29}

31fn format_uptime(value: Duration) -> anyhow::Result<String> {
32    let printer = jiff::fmt::friendly::SpanPrinter::new()
33        .spacing(jiff::fmt::friendly::Spacing::BetweenUnitsAndDesignators)
34        .designator(jiff::fmt::friendly::Designator::Verbose);
35
36    let now = jiff::Timestamp::now().to_zoned(jiff::tz::TimeZone::UTC);
37    let span = jiff::Span::try_from(value).context("Failed to create span")?;
38    let round_options = jiff::SpanRound::new()
39        .smallest(jiff::Unit::Second)
40        .largest(jiff::Unit::Year)
41        .relative(&now);
42    let span_rounded = span.round(round_options).context("Failed to round")?;
43
44    Ok(printer.span_to_string(&span_rounded))
45}

47fn format_cpu_frequency(hz: u64) -> String {
48    const HZ_PER_GHZ: f64 = 1_000_000_000.0;
49
50    let hz = hz as f64;
51    let ghz = hz / HZ_PER_GHZ;
52
53    format!("{ghz:.2} GHz")
54}

56fn format_memory_usage(total: u64, available: u64) -> String {
57    const BYTES_PER_GB: f64 = 1024.0 * 1024.0 * 1024.0;
58
59    let used_gib = (total - available) as f64 / BYTES_PER_GB;
60    let total_gib = total as f64 / BYTES_PER_GB;
61
62    format!("{used_gib:.2} GiB / {total_gib:.2} GiB",)
63}

84    async fn new() -> Self {
85        let cache_context = Arc::new(pikadick_system_info::CacheContext::new());
86
87        let hostname = match cache_context
88            .get_hostname()
89            .context("Failed to get hostname")
90        {
91            Ok(hostname) => Some(hostname),
92            Err(error) => {
93                warn!("{error:?}");
94                None
95            }
96        };
97
98        let system_name = match cache_context
99            .get_system_name()
100            .context("Failed to get system name")
101        {
102            Ok(system_name) => system_name,
103            Err(error) => {
104                warn!("{error:?}");
105                None
106            }
107        };
108
109        let system_version = match cache_context
110            .get_system_version()
111            .context("Failed to get system version")
112        {
113            Ok(system_name) => Some(system_name),
114            Err(error) => {
115                warn!("{error:?}");
116                None
117            }
118        };
119
120        let architecture = match cache_context
121            .get_architecture()
122            .context("Failed to get architecture")
123        {
124            Ok(architecture) => Some(
125                architecture
126                    .map(|architecture| architecture.as_str())
127                    .unwrap_or("Unknown"),
128            ),
129            Err(error) => {
130                warn!("{error:?}");
131                None
132            }
133        };
134
135        let boot_time = match cache_context
136            .get_boot_time()
137            .context("Failed to get boot time")
138            .and_then(format_boot_time)
139        {
140            Ok(boot_time) => Some(boot_time),
141            Err(error) => {
142                warn!("{error:?}");
143                None
144            }
145        };
146
147        let uptime = match cache_context
148            .get_uptime()
149            .context("Failed to get uptime")
150            .and_then(format_uptime)
151        {
152            Ok(uptime) => Some(uptime),
153            Err(error) => {
154                warn!("{error:?}");
155                None
156            }
157        };
158
159        let result = {
160            let cache_context = cache_context.clone();
161            tokio::task::spawn_blocking(move || {
162                cache_context.start()?;
163                std::thread::sleep(Duration::from_secs(1));
164                cache_context.end()?;
165                anyhow::Ok(())
166            })
167            .await
168            .map_err(anyhow::Error::from)
169            .flatten()
170            .context("Failed to collect timed stats")
171            .map_err(ArcAnyhowError::new)
172        };
173
174        let cpu_frequency = match result
175            .as_ref()
176            .map_err(|value| value.clone())
177            .and_then(|_| {
178                cache_context
179                    .get_cpu_frequency()
180                    .map_err(anyhow::Error::from)
181                    .map_err(ArcAnyhowError::new)
182            })
183            .map(format_cpu_frequency)
184            .context("Failed to get cpu frequency")
185        {
186            Ok(value) => Some(value),
187            Err(error) => {
188                warn!("{error:?}");
189                None
190            }
191        };
192
193        let min_cpu_frequency = match cache_context
194            .get_min_cpu_frequency()
195            .context("Failed to get min cpu frequency")
196        {
197            Ok(value) => Some(format_cpu_frequency(value)),
198            // Don't warn for unsupported APIs; too noisy.
199            Err(error)
200                if error
201                    .root_cause()
202                    .downcast_ref::<pikadick_system_info::Error>()
203                    .is_some_and(|error| error.is_unsupported()) =>
204            {
205                None
206            }
207            Err(error) => {
208                warn!("{error:?}");
209                None
210            }
211        };
212
213        // This is an underestimate on Windows;
214        // there does not seem to be an API for this that includes TurboBoost.
215        let max_cpu_frequency = match cache_context
216            .get_max_cpu_frequency()
217            .context("Failed to get max CPU frequency")
218        {
219            Ok(value) => Some(format_cpu_frequency(value)),
220            Err(error) => {
221                warn!("{error:?}");
222                None
223            }
224        };
225
226        let cpu_logical_count = match cache_context
227            .count_logical_cpus()
228            .context("Failed to count number of logical cpus")
229        {
230            Ok(cpu_logical_count) => Some(cpu_logical_count),
231            Err(error) => {
232                warn!("{error:?}");
233                None
234            }
235        };
236
237        let cpu_physical_count = match cache_context
238            .count_physical_cpus()
239            .context("Failed to count number of physical cpus")
240        {
241            Ok(cpu_physical_count) => Some(cpu_physical_count),
242            Err(error) => {
243                warn!("{error:?}");
244                None
245            }
246        };
247
248        let cpu_core_count = match (cpu_logical_count, cpu_physical_count) {
249            (Some(logical_count), Some(physical_count)) => Some(format!(
250                "{logical_count} logical, {physical_count} physical"
251            )),
252            (Some(logical_count), None) => Some(format!("{logical_count} logical")),
253            (None, Some(physical_count)) => Some(format!("{physical_count} physical")),
254            (None, None) => None,
255        };
256
257        let total_memory = match cache_context
258            .get_total_memory()
259            .context("Failed to get total memory")
260        {
261            Ok(memory) => Some(memory),
262            Err(error) => {
263                warn!("{error:?}");
264                None
265            }
266        };
267
268        let available_memory = match cache_context
269            .get_available_memory()
270            .context("Failed to get available memory")
271        {
272            Ok(memory) => Some(memory),
273            Err(error) => {
274                warn!("{error:?}");
275                None
276            }
277        };
278
279        let memory_usage = match (total_memory, available_memory) {
280            (Some(total_memory), Some(available_memory)) => {
281                Some(format_memory_usage(total_memory, available_memory))
282            }
283            _ => None,
284        };
285
286        let total_swap = match cache_context
287            .get_total_swap()
288            .context("Failed to get total swap")
289        {
290            Ok(memory) => Some(memory),
291            Err(error) => {
292                warn!("{error:?}");
293                None
294            }
295        };
296
297        let available_swap = match cache_context
298            .get_available_swap()
299            .context("Failed to get available swap")
300        {
301            Ok(memory) => Some(memory),
302            Err(error) => {
303                warn!("{error:?}");
304                None
305            }
306        };
307
308        let swap_usage = match (total_swap, available_swap) {
309            (Some(total_swap), Some(available_swap)) => {
310                Some(format_memory_usage(total_swap, available_swap))
311            }
312            _ => None,
313        };
314
315        let cpu_usage = match cache_context
316            .get_cpu_usage()
317            .context("Failed to get CPU usage")
318            .map(|cpu_usage| format!("{cpu_usage:.2}%"))
319        {
320            Ok(cpu_usage) => Some(cpu_usage),
321            Err(error) => {
322                warn!("{error:?}");
323                None
324            }
325        };
326
327        let cpu_temp = {
328            let cache_context = cache_context.clone();
329            let result = tokio::task::spawn_blocking(move || {
330                cache_context
331                    .get_cpu_temperature()
332                    .map_err(anyhow::Error::from)
333            })
334            .await
335            .map_err(anyhow::Error::from)
336            .flatten()
337            .context("Failed to get CPU temp")
338            .map(|value| format!("{value:.2} °C"));
339            match result {
340                Ok(cpu_temp) => Some(cpu_temp),
341                // Don't warn for unsupported APIs; too noisy.
342                Err(error)
343                    if error
344                        .root_cause()
345                        .downcast_ref::<pikadick_system_info::Error>()
346                        .is_some_and(|error| error.is_unsupported()) =>
347                {
348                    None
349                }
350                Err(error) => {
351                    warn!("{error:?}");
352                    None
353                }
354            }
355        };
356
357        Self {
358            hostname,
359            system_name,
360            system_version,
361            architecture,
362            boot_time,
363            uptime,
364            cpu_frequency,
365            min_cpu_frequency,
366            max_cpu_frequency,
367            cpu_core_count,
368            memory_usage,
369            swap_usage,
370            cpu_usage,
371            cpu_temp,
372        }
373    }

375    fn populate_embed(&self, mut create_embed: CreateEmbed) -> CreateEmbed {
376        let fields = [
377            ("Hostname", self.hostname.as_deref()),
378            ("OS", self.system_name.as_deref()),
379            ("OS Version", self.system_version.as_deref()),
380            ("Architecture", self.architecture),
381            ("Boot Time", self.boot_time.as_deref()),
382            ("Uptime", self.uptime.as_deref()),
383            ("CPU Freq", self.cpu_frequency.as_deref()),
384            ("Min CPU Freq", self.min_cpu_frequency.as_deref()),
385            ("Max CPU Freq", self.max_cpu_frequency.as_deref()),
386            ("Cpu Core Count", self.cpu_core_count.as_deref()),
387            ("Memory Usage", self.memory_usage.as_deref()),
388            ("Swap Usage", self.swap_usage.as_deref()),
389            ("Cpu Usage", self.cpu_usage.as_deref()),
390            ("Cpu Temp", self.cpu_temp.as_deref()),
391        ];
392
393        for (key, value) in fields {
394            if let Some(value) = value {
395                create_embed = create_embed.field(key, value, true);
396            }
397        }
398
399        create_embed
400    }
401}

403#[poise::command(slash_command, description_localized("en-US", "Get system stats."))]
404pub async fn system(ctx: PoiseContext<'_>) -> Result<(), PoiseError> {
405    ctx.defer().await?;
406
407    let start = Instant::now();
408    let profile_avatar_url = ctx.cache().current_user().avatar_url();
409    let system_stats = SystemStats::new().await;
410    let data_retrieval_time = start.elapsed();
411
412    let mut create_embed = CreateEmbed::new()
413        .title("System Status")
414        .color(Colour::from_rgb(255, 0, 0));
415    if let Some(icon) = profile_avatar_url {
416        create_embed = create_embed.thumbnail(icon);
417    }
418
419    create_embed = system_stats.populate_embed(create_embed);
420
421    let footer = CreateEmbedFooter::new(format!(
422        "Retrieved system data in {:.2} second(s)",
423        data_retrieval_time.as_secs_f32()
424    ));
425    create_embed = create_embed.footer(footer);
426
427    let create_reply = CreateReply::new().embed(create_embed);
428    ctx.send(create_reply).await?;
429
430    Ok(())
431}
