60/// Calculate the target bitrate.
61///
62/// `target_size` is in kilobits.
63///
64/// # Returns
65/// Returns the bitrate in kilobits.
66fn calc_target_bitrate(target_size: u64, duration: Duration) -> u64 {
67    // https://stackoverflow.com/questions/29082422/ffmpeg-video-compression-specific-file-size
68
69    target_size / duration.as_secs()
70}

77/// Parse a tiktok url.
78fn parse_tiktok_url(url: &Url) -> anyhow::Result<UrlData> {
79    let host_str = url.host_str().context("Missing host")?;
80    ensure!(host_str == "www.tiktok.com" || host_str == "tiktok.com");
81
82    let mut path_segments = url.path_segments().context("Missing path")?;
83    let _user = path_segments.next().context("Missing user")?;
84    ensure!(path_segments.next() == Some("video"));
85    let id = path_segments
86        .next()
87        .context("Missing id")?
88        .parse()
89        .context("Failed to parse id")?;
90
91    Ok(UrlData { id })
92}

94async fn select_best_encoder(encoder_task: &EncoderTask) -> anyhow::Result<&'static str> {
95    let mut encoders = encoder_task
96        .get_encoders(true)
97        .await
98        .context("Failed to get encoders")?;
99
100    // Keep only h264 encoders
101    encoders.retain(|encoder| encoder.description.ends_with("(codec h264)"));
102    debug!("Found h264 encoders: {encoders:#?}");
103
104    let mut best_encoder_index = None;
105    for encoder in encoders {
106        let encoder_index = ENCODER_PREFERENCE_LIST
107            .iter()
108            .position(|name| **name == *encoder.name);
109        let encoder_index = match encoder_index {
110            Some(encoder_index) => encoder_index,
111            None => continue,
112        };
113        if best_encoder_index.is_none_or(|best_encoder_index| best_encoder_index > encoder_index) {
114            best_encoder_index = Some(encoder_index);
115        }
116    }
117
118    let best_encoder_index = best_encoder_index.context("Failed to select an encoder")?;
119    let best_encoder = ENCODER_PREFERENCE_LIST[best_encoder_index];
120
121    Ok(best_encoder)
122}

150    pub async fn new<P>(cache_dir: P, encoder_task: EncoderTask) -> anyhow::Result<Self>
151    where
152        P: AsRef<Utf8Path>,
153    {
154        let cache_dir = cache_dir.as_ref();
155        let video_download_cache_path = cache_dir.join("tiktok");
156
157        // TODO: Expand into proper filecache manager
158        tokio::fs::create_dir_all(&video_download_cache_path)
159            .await
160            .context("Failed to create tiktok cache dir")?;
161
162        let best_encoder = select_best_encoder(&encoder_task).await?;
163        info!("Selected encoder \"{best_encoder}\"");
164
165        Ok(Self {
166            inner: Arc::new(InnerTikTokData {
167                tikwm_client: tikwm::Client::new(),
168                encoder_task,
169
170                video_download_cache_path,
171                video_download_request_map: RequestMap::new(),
172                video_encoder: best_encoder,
173            }),
174        })
175    }

177    async fn create_reencoded_file(&self, id: u64) -> anyhow::Result<Utf8PathBuf> {
178        let client = self.inner.tikwm_client.client.clone();
179        let video_encoder = self.inner.video_encoder;
180
181        let reencoded_file_name = format!("{id}-reencoded.mp4");
182        let reencoded_file_path = self
183            .inner
184            .video_download_cache_path
185            .join(reencoded_file_name);
186
187        let file_name = format!("{id}.mp4");
188        let file_path = self.inner.video_download_cache_path.join(file_name);
189
190        if tokio::fs::try_exists(&reencoded_file_path)
191            .await
192            .context("Failed to get metadata of re-encoded file")?
193        {
194            // The reencoded file is present. Use it.
195            return Ok(reencoded_file_path);
196        }
197
198        // Get the metadata of the raw file.
199        // Download it if needed.
200        let metadata = match crate::util::try_metadata(&file_path).await {
201            Ok(Some(metadata)) => metadata,
202            Ok(None) => {
203                // File not present. Download it.
204                info!("Downloading tiktok video {id:?} from tikwm");
205
206                let create_download_response = self
207                    .inner
208                    .tikwm_client
209                    .create_download_task(itoa::Buffer::new().format(id))
210                    .await
211                    .context("Failed to create video download")?;
212                ensure!(create_download_response.id == id);
213
214                let download_task_result = self
215                    .inner
216                    .tikwm_client
217                    .get_task_result(&create_download_response.task_id)
218                    .await
219                    .context("Failed to create video download")?;
220
221                info!(
222                    "Downloading tikwm from {:?}",
223                    download_task_result.download_url.as_str()
224                );
225                ensure!(
226                    download_task_result
227                        .download_url
228                        .path_segments()
229                        .and_then(|mut path| path.next_back()?.rsplit_once('.'))
230                        .is_some_and(|(_stem, extension)| extension == "mp4")
231                );
232
233                nd_util::download_to_path(
234                    &client,
235                    download_task_result.download_url.as_str(),
236                    &file_path,
237                )
238                .await?;
239                tokio::fs::metadata(&file_path)
240                    .await
241                    .context("Failed to get file metadata")?
242            }
243            Err(e) => {
244                return Err(e).context("Failed to get metadata of file");
245            }
246        };
247
248        // If the file is less than than 8mb, we don't need to re-encode it
249        if metadata.len() < FILE_SIZE_LIMIT_BYTES {
250            return Ok(file_path);
251        }
252
253        let ffprobe_result = tokio_ffmpeg_cli::probe(file_path.as_str())
254            .await
255            .context("Failed to ffprobe")?;
256        let video_duration = Duration::from_secs_f64(ffprobe_result.format.duration);
257
258        // We target half of the maximum size to give ourselves some lee-way.
259        // This merely sets the target bit-rate, and we don't take into account audio size.
260        let target_bitrate =
261            calc_target_bitrate((TARGET_FILE_SIZE_BYTES / 1024) * 8 / 2, video_duration);
262        let reencoded_file_path_tmp_1 =
263            DropRemovePath::new(reencoded_file_path.with_extension("1.tmp"));
264
265        info!(
266            "Re-encoding tiktok video {:?} to {:?} @ video bitrate {}",
267            file_path,
268            reencoded_file_path_tmp_1.display(),
269            target_bitrate
270        );
271
272        {
273            let mut stream = self
274                .inner
275                .encoder_task
276                .encode()
277                .input(&file_path)
278                .output(&*reencoded_file_path_tmp_1)
279                .audio_codec("copy")
280                .video_codec(video_encoder)
281                .video_bitrate(format!("{target_bitrate}K"))
282                .output_format("mp4")
283                .try_send()
284                .await
285                .context("Failed to start re-encoding")?;
286
287            let mut maybe_exit_status = None;
288            while let Some(msg) = stream.next().await {
289                match msg.context("FFMpeg stream error") {
290                    Ok(tokio_ffmpeg_cli::Event::ExitStatus(exit_status)) => {
291                        maybe_exit_status = Some(exit_status);
292                    }
293                    Ok(tokio_ffmpeg_cli::Event::Progress(_progress)) => {
294                        // For now, we don't care about progress as there is no way to report it to the user on discord.
295                    }
296                    Ok(tokio_ffmpeg_cli::Event::Unknown(_line)) => {
297                        // warn!("unknown ffmpeg line: `{}`", line);
298                        // We don't care about unkown lines
299                    }
300                    Err(error) => {
301                        warn!("{error:?}");
302                    }
303                }
304            }
305
306            let exit_status = maybe_exit_status.context("Stream did not report an exit status")?;
307
308            // Validate exit status
309            ensure!(exit_status.success(), "Invalid exit status");
310        }
311
312        // The RPI's ffmpeg produces invalid mp4 files.
313        // Until we can investigate and fix, transcode the file to try to let ffmpeg fix it.
314        let reencoded_file_path_tmp_2 =
315            DropRemovePath::new(reencoded_file_path.with_extension("2.tmp"));
316
317        {
318            let mut stream = self
319                .inner
320                .encoder_task
321                .encode()
322                .input(&*reencoded_file_path_tmp_1)
323                .output(&*reencoded_file_path_tmp_2)
324                .audio_codec("copy")
325                .video_codec("copy")
326                .output_format("mp4")
327                .try_send()
328                .await
329                .context("failed to start transcoding")?;
330
331            let mut maybe_exit_status = None;
332            while let Some(msg) = stream.next().await {
333                match msg.context("ffmpeg stream error") {
334                    Ok(tokio_ffmpeg_cli::Event::ExitStatus(exit_status)) => {
335                        maybe_exit_status = Some(exit_status);
336                    }
337                    Ok(tokio_ffmpeg_cli::Event::Progress(_progress)) => {
338                        // For now, we don't care about progress as there is no way to report it to the user on discord.
339                    }
340                    Ok(tokio_ffmpeg_cli::Event::Unknown(_line)) => {
341                        // warn!("unknown ffmpeg line: `{}`", line);
342                        // We don't care about unkown lines
343                    }
344                    Err(error) => {
345                        warn!("{error:?}");
346                    }
347                }
348            }
349
350            let exit_status = maybe_exit_status.context("Stream did not report an exit status")?;
351
352            // Validate exit status
353            ensure!(exit_status.success(), "Invalid exit status");
354        }
355
356        let mut reencoded_file_path_tmp = reencoded_file_path_tmp_2;
357
358        // Validate file size
359        let metadata = tokio::fs::metadata(&reencoded_file_path_tmp)
360            .await
361            .context("Failed to get metadata of encoded file")?;
362        let metadata_len = metadata.len();
363        ensure!(
364            metadata_len < FILE_SIZE_LIMIT_BYTES,
365            "Re-encoded file size ({metadata_len}) is larger than the limit {FILE_SIZE_LIMIT_BYTES}",
366        );
367
368        // Rename the tmp file to be the actual name.
369        tokio::fs::rename(&*reencoded_file_path_tmp, &reencoded_file_path)
370            .await
371            .context("Failed to rename temp file")?;
372
373        // "Persist" the tmp file, as in don't try to remove it
374        reencoded_file_path_tmp.persist();
375
376        Ok(reencoded_file_path)
377    }

379    /// Get video data, using the cache if needed.
380    pub async fn get_video_data_cached(&self, id: u64) -> anyhow::Result<Arc<Utf8Path>> {
381        self.inner
382            .video_download_request_map
383            .get_or_fetch(id.to_string(), || {
384                let self_clone = self.clone();
385
386                async move {
387                    self_clone
388                        .create_reencoded_file(id)
389                        .await
390                        .map(Arc::from)
391                        .map_err(ArcAnyhowError::new)
392                }
393            })
394            .await
395            .map_err(From::from)
396    }

398    /// Try embedding a url.
399    pub async fn try_embed_url(
400        &self,
401        ctx: &Context,
402        msg: &Message,
403        url: &Url,
404        loading_reaction: &mut Option<LoadingReaction>,
405        delete_link: bool,
406    ) -> anyhow::Result<()> {
407        let url_data = parse_tiktok_url(url)?;
408
409        let video_path = self
410            .get_video_data_cached(url_data.id)
411            .await
412            .context("Failed to download tiktok video")?;
413
414        let file = CreateAttachment::path(video_path.as_std_path()).await?;
415        let message_builder = CreateMessage::new().add_file(file);
416        msg.channel_id
417            .send_message(&ctx.http, message_builder)
418            .await?;
419
420        if let Some(mut loading_reaction) = loading_reaction.take() {
421            loading_reaction.send_ok();
422
423            if delete_link {
424                msg.delete(&ctx.http)
425                    .await
426                    .context("Failed to delete original message")?;
427            }
428        }
429
430        Ok(())
431    }

439#[poise::command(
440    slash_command,
441    rename = "tiktok-embed",
442    description_localized("en-US", "Configure tiktok embeds for this server"),
443    default_member_permissions = "ADMINISTRATOR"
444)]
445pub async fn tiktok_embed(
446    ctx: PoiseContext<'_>,
447    #[description = "Whether embeds should be enabled for this server"] enable: Option<bool>,
448    #[description = "Whether source messages should be deleted"]
449    #[rename = "delete-link"]
450    delete_link: Option<bool>,
451) -> Result<(), PoiseError> {
452    let data_lock = ctx.serenity_context().data.read().await;
453    let client_data = data_lock.get::<ClientDataKey>().unwrap();
454    let db = client_data.db.clone();
455    drop(data_lock);
456
457    let guild_id = match ctx.guild_id() {
458        Some(id) => id,
459        None => {
460            ctx.reply("Missing server id. Are you in a server right now?")
461                .await?;
462            return Ok(());
463        }
464    };
465
466    let mut set_flags = TikTokEmbedFlags::empty();
467    let mut unset_flags = TikTokEmbedFlags::empty();
468    if let Some(enable) = enable {
469        if enable {
470            set_flags.insert(TikTokEmbedFlags::ENABLED);
471        } else {
472            unset_flags.insert(TikTokEmbedFlags::ENABLED);
473        }
474    }
475    if let Some(enable) = delete_link {
476        if enable {
477            set_flags.insert(TikTokEmbedFlags::DELETE_LINK);
478        } else {
479            unset_flags.insert(TikTokEmbedFlags::DELETE_LINK);
480        }
481    }
482
483    let flags = if set_flags.is_empty() && unset_flags.is_empty() {
484        db.get_tiktok_embed_flags(guild_id).await?
485    } else {
486        let (_old_flags, new_flags) = db
487            .set_tiktok_embed_flags(guild_id, set_flags, unset_flags)
488            .await?;
489        new_flags
490    };
491
492    let embed_builder = CreateEmbed::new()
493        .title("TikTok Embeds")
494        .field(
495            "Enabled?",
496            bool_to_str(flags.contains(TikTokEmbedFlags::ENABLED)),
497            false,
498        )
499        .field(
500            "Delete link?",
501            bool_to_str(flags.contains(TikTokEmbedFlags::DELETE_LINK)),
502            false,
503        );
504
505    ctx.send(CreateReply::default().reply(true).embed(embed_builder))
506        .await?;
507
508    Ok(())
509}
