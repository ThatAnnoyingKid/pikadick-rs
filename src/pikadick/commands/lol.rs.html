77    /// Get a player profile suggestion.

79        info!("Searching for lol user with \"{user}\"");
80
81        let mut users = self
82            .client
83            .search_users(user)
84            .await
85            .context("Failed to search for users")?;
86        ensure!(!users.is_empty(), "Failed to find user from \"{user}\"");

109    async fn download_to_path(&self, url: &str, path: &Utf8Path) -> anyhow::Result<()> {
110        let temp_path = path.with_added_extension("temp");
111        let mut file = tokio::fs::File::create(&temp_path).await?;
112        let mut response = self
113            .client
114            .client
115            .get(url)
116            .send()
117            .await?
118            .error_for_status()?;
119        while let Some(chunk) = response.chunk().await? {
120            file.write_all(&chunk).await?;
121        }
122        file.flush().await?;
123        file.sync_all().await?;
124
125        tokio::fs::rename(temp_path, &path).await?;
126
127        anyhow::Ok(())
128    }

130    pub async fn get_icon_path(&self, icon_id: u32) -> anyhow::Result<Utf8PathBuf> {
131        let lock = self.icon_lock_map.lock(icon_id).await;
132        let path = self.profile_icons_dir.join(format!("{icon_id}.png"));
133
134        if tokio::fs::try_exists(&path).await? {
135            return Ok(path);
136        }
137
138        info!("Downloading lol profile icon {icon_id}");
139
140        let url = format!(
141            "https://static.bigbrain.gg/assets/lol/riot_static/16.22.1/img/profileicon/{icon_id}.png"
142        );
143        self.download_to_path(url.as_str(), &path).await?;
144
145        drop(lock);
146
147        Ok(path)
148    }

171    pub async fn render_user_stats(&self, user: ugg::model::User) -> anyhow::Result<Vec<u8>> {
172        let icon_path = self.get_icon_path(user.icon_id).await?;
173        let icon = tokio::fs::read(icon_path).await?;
174
175        let rank_icon = match user.tier.as_ref() {
176            Some(tier) => {
177                let path = self.get_rank_icon_path(tier).await?;
178                Some(tokio::fs::read(path).await?)
179            }
180            None => None,
181        };
182
183        let mut renderer = self.user_stats_renderer.clone().lock_owned().await;
184        let png = tokio::task::spawn_blocking(move || {
185            let start = Instant::now();
186            let png = renderer.render(user, icon, rank_icon)?;
187            let elapsed = start.elapsed();
188
189            info!("Rendered lol user stats in {elapsed:?}");
190
191            anyhow::Ok(png)
192        })
193        .await??;
194
195        Ok(png)
196    }

199async fn autocomplete_user(ctx: PoiseContext<'_>, partial: &str) -> CreateAutocompleteResponse {
200    if partial.is_empty() {
201        return CreateAutocompleteResponse::new();
202    }
203
204    let ugg_client = &ctx.data().ugg_client;
205
206    let users = match ugg_client.search_users(partial).await {
207        Ok(users) => users,
208        Err(error) => {
209            warn!("{error:?}");
210            return CreateAutocompleteResponse::new();
211        }
212    };
213    let choices = users
214        .iter()
215        .map(|user| format!("{}#{}", user.riot_user_name, user.riot_tag_line).into())
216        .collect();
217
218    CreateAutocompleteResponse::new().set_choices(choices)
219}
