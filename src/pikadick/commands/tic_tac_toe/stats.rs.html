1use crate::{
2    PoiseContext,
3    PoiseError,
4    util::AsciiTable,
5};
6use anyhow::Context as _;
7use tracing::error;
8
9#[poise::command(
10    slash_command,
11    description_localized("en-US", "Get personal stats for Tic-Tac-Toe.")
12)]
13pub async fn stats(ctx: PoiseContext<'_>) -> Result<(), PoiseError> {
14    let guild_id = ctx.guild_id();
15    let author = ctx.author();
16    let author_id = author.id;
17
18    let scores = match ctx
19        .data()
20        .tic_tac_toe_data
21        .get_score(guild_id, author_id)
22        .await
23        .context("Failed to get tic-tac-toe stats")
24    {
25        Ok(scores) => scores,
26        Err(error) => {
27            error!("{error:?}");
28            ctx.reply(format!("{error:?}")).await?;
29            return Ok(());
30        }
31    };
32
33    let mut table = AsciiTable::new(4, 2);
34    table.set_padding(2);
35
36    let mut wins_buffer = itoa::Buffer::new();
37    let mut losses_buffer = itoa::Buffer::new();
38    let mut ties_buffer = itoa::Buffer::new();
39    let mut concedes_buffer = itoa::Buffer::new();
40
41    table.set_cell(0, 0, "Wins");
42    table.set_cell(1, 0, "Losses");
43    table.set_cell(2, 0, "Ties");
44    table.set_cell(3, 0, "Concedes");
45
46    table.set_cell(0, 1, wins_buffer.format(scores.wins));
47    table.set_cell(1, 1, losses_buffer.format(scores.losses));
48    table.set_cell(2, 1, ties_buffer.format(scores.ties));
49    table.set_cell(3, 1, concedes_buffer.format(scores.concedes));
50
51    let response = format!("```\n{}'s Tic-Tac-Toe Stats\n{}\n```", author.name, table);
52    ctx.reply(response).await?;
53
54    Ok(())
55}
