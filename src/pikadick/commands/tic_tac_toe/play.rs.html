18#[derive(Debug, poise::ChoiceParameter)]
19enum TeamChoice {
20    #[name = "X"]
21    X,
22
23    #[name = "O"]
24    O,
25}
26
27#[poise::command(
28    slash_command,
29    description_localized("en-US", "Start a game of Tic-Tac-Toe.")
30)]
31pub async fn play(
32    ctx: PoiseContext<'_>,
33    #[description_localized("en-US", "The user you want to face. Omit to face the computer.")]
34    opponent: Option<UserId>,
35    #[description_localized("en-US", "The team you want to play as, X or O. Defaults to X.")] team: Option<
36        TeamChoice,
37    >,
38) -> Result<(), PoiseError> {
39    let author = ctx.author();
40    let author_id = author.id;
41    let author_team = match team {
42        Some(TeamChoice::X) | None => tic_tac_toe::Team::X,
43        Some(TeamChoice::O) => tic_tac_toe::Team::O,
44    };
45    let opponent = match opponent {
46        Some(user_id) => TicTacToePlayer::User(user_id),
47        None => TicTacToePlayer::Computer,
48    };
49    let create_game_result = ctx
50        .data()
51        .tic_tac_toe_data
52        .create_game(ctx.guild_id(), author_id, author_team, opponent)
53        .await;
54    let game = match create_game_result {
55        Ok(game) => game,
56        Err(TicTacToeCreateGameError::AuthorInGame) => {
57            let response = "Finish your current game in this server before starting a new one. Use `tic-tac-toe concede` to end your current game.";
58            ctx.reply(response).await?;
59            return Ok(());
60        }
61        Err(TicTacToeCreateGameError::OpponentInGame) => {
62            let response = "Your opponent is currently in another game in this server. Wait for them to finish.";
63            ctx.reply(response).await?;
64            return Ok(());
65        }
66        Err(TicTacToeCreateGameError::Database(error)) => {
67            error!("{error:?}");
68            ctx.reply("Database error").await?;
69            return Ok(());
70        }
71    };
72
73    let game_board = game.board;
74    let user = if let TicTacToePlayer::User(opponent_id) = opponent {
75        // Cannot be a computer here as there are at least 2 human players at this point
76        if author_team == tic_tac_toe::Team::X {
77            author_id
78        } else {
79            opponent_id
80        }
81    } else {
82        // The opponent is not a user, so it is a computer.
83        // We already calculated the move and updated if the computer is X.
84        // All that's left is to @author and print the board state.
85        author_id
86    };
87
88    let file = match ctx
89        .data()
90        .tic_tac_toe_data
91        .render_board(game_board)
92        .await
93        .context("Failed to render Tic-Tac-Toe board")
94    {
95        Ok(file) => CreateAttachment::bytes(file, format!("{}.png", game_board.encode_u16())),
96        Err(error) => {
97            error!("{error:?}");
98            ctx.reply(format!("{error:?}")).await?;
99            return Ok(());
100        }
101    };
102
103    let create_reply = CreateReply::new()
104        .content(format!("Game created! Your turn {}", user.mention()))
105        .attachment(file);
106
107    ctx.send(create_reply).await?;
108
109    Ok(())
110}
