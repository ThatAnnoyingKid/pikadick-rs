1use crate::{
2    PoiseContext,
3    PoiseError,
4    database::model::TicTacToePlayer,
5};
6use anyhow::Context;
7use poise::CreateReply;
8use serenity::{
9    builder::CreateAttachment,
10    prelude::Mentionable,
11};
12use tracing::error;
13
14#[poise::command(
15    slash_command,
16    description_localized("en-US", "Concede a game of Tic-Tac-Toe.")
17)]
18pub async fn concede(ctx: PoiseContext<'_>) -> Result<(), PoiseError> {
19    let tic_tac_toe_data = &ctx.data().tic_tac_toe_data;
20
21    let guild_id = ctx.guild_id();
22    let author_id = ctx.author().id;
23
24    let game = match tic_tac_toe_data.concede_game(guild_id, author_id).await {
25        Ok(Some(game)) => game,
26        Ok(None) => {
27            let response = "Failed to concede as you have no games in this server";
28            ctx.reply(response).await?;
29            return Ok(());
30        }
31        Err(error) => {
32            error!("{error:?}");
33            ctx.reply("Database error").await?;
34            return Ok(());
35        }
36    };
37
38    let opponent = game
39        .get_opponent(TicTacToePlayer::User(author_id))
40        .expect("Author is not playing the game");
41
42    let file = match tic_tac_toe_data
43        .renderer
44        .render_board_async(game.board)
45        .await
46        .context("Failed to render Tic-Tac-Toe board")
47    {
48        Ok(file) => CreateAttachment::bytes(file, format!("ttt-{}.png", game.board.encode_u16())),
49        Err(error) => {
50            error!("{error:?}");
51            ctx.reply(format!("{error:?}")).await?;
52            return Ok(());
53        }
54    };
55
56    let content = format!(
57        "{} has conceded to {}.",
58        author_id.mention(),
59        opponent.mention()
60    );
61
62    let create_reply = CreateReply::new().content(content).attachment(file);
63    ctx.send(create_reply).await?;
64
65    Ok(())
66}
