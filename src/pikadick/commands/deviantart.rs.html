26const DATA_STORE_NAME: &str = "deviantart";
27const COOKIE_KEY: &str = "cookie-store";
28
29const FIVE_MINUTES: Duration = Duration::from_secs(60 * 5);
30
31/// A caching deviantart client
32#[derive(Debug)]
33pub struct DeviantartClient {
34    username: String,
35    password: String,
36    client: deviantart::Client,
37    database: Database,
38    search_cache: AsyncTimedLruCache<String, Result<Arc<[Deviation]>, ArcAnyhowError>>,
39}
40
41impl DeviantartClient {
42    /// Make a new [`DeviantartClient`].
43    pub async fn new(config: &Config, database: Database) -> anyhow::Result<Self> {
44        let client = deviantart::Client::new();
45
46        let cookie_data: Option<Vec<u8>> = database
47            .store_get(DATA_STORE_NAME, COOKIE_KEY)
48            .await
49            .context("Failed to get cookie data")?;
50
51        match cookie_data {
52            Some(cookie_data) => {
53                // Ignore cookies if they fail to parse.
54                if let Err(error) = client
55                    .load_json_cookies(BufReader::new(std::io::Cursor::new(cookie_data)))
56                    .await
57                    .context("Failed to load JSON cookies")
58                {
59                    warn!("{error:?}");
60                }
61            }
62            None => {
63                info!("Could not load cookie data");
64            }
65        }
66
67        let search_cache = AsyncTimedLruCache::new(100, FIVE_MINUTES);
68
69        Ok(DeviantartClient {
70            username: config.deviantart.username.clone(),
71            password: config.deviantart.password.clone(),
72            client,
73            database,
74            search_cache,
75        })
76    }
77
78    /// Signs in if necessary
79    pub async fn sign_in(&self) -> anyhow::Result<()> {
80        if !self.client.is_logged_in_online().await? {
81            info!("Re-signing in");
82            self.client.login(&self.username, &self.password).await?;
83
84            let cookie_data = Vec::with_capacity(2 * 1024);
85            let cookie_data = self
86                .client
87                .save_json_cookies(cookie_data)
88                .await
89                .context("Failed to save JSON cookies")?;
90            self.database
91                .store_put(DATA_STORE_NAME, COOKIE_KEY, &cookie_data)
92                .await?;
93        }
94
95        Ok(())
96    }
97
98    /// Search for deviantart images with a cache.
99    pub async fn search(&self, query: &str) -> Result<Arc<[Deviation]>, ArcAnyhowError> {
100        self.search_cache
101            .get(query.to_string(), || async {
102                let start = Instant::now();
103                let result = async {
104                    self.sign_in()
105                        .await
106                        .context("Failed to log in to deviantart")?;
107
108                    let mut search_cursor = self.client.search(query, None);
109                    search_cursor
110                        .next_page()
111                        .await
112                        .context("Failed to search")?;
113
114                    let list = search_cursor
115                        .take_current_deviations()
116                        .expect("Missing page")
117                        .context("Failed to process results")?;
118
119                    anyhow::Ok(list)
120                }
121                .await
122                .map(Arc::from)
123                .map_err(ArcAnyhowError::new);
124
125                let elapsed = start.elapsed();
126                info!("Searched deviantart in {elapsed:?}");
127
128                result
129            })
130            .await
131    }
132}

134#[poise::command(
135    slash_command,
136    description_localized("en-US", "Search for art on deviantart")
137)]
138pub async fn deviantart(
139    ctx: PoiseContext<'_>,
140    #[description = "The deviantart search query."] query: String,
141) -> Result<(), PoiseError> {
142    ctx.defer().await?;
143
144    info!("Searching for \"{query}\" on deviantart");
145    let result = ctx.data().deviantart_client.search(&query).await;
146    match result {
147        Ok(data) => {
148            let choice = data
149                .iter()
150                .filter_map(|deviation| {
151                    if deviation.is_image() {
152                        Some(
153                            deviation
154                                .get_image_download_url()
155                                .or_else(|| deviation.get_fullview_url(Default::default()).ok()),
156                        )
157                    } else if deviation.is_film() {
158                        Some(deviation.get_best_video_url().cloned())
159                    } else {
160                        None
161                    }
162                })
163                .choose(&mut rand::rng());
164
165            if let Some(choice) = choice {
166                if let Some(url) = choice {
167                    ctx.reply(url).await?;
168                } else {
169                    ctx.reply("Missing url. This is probably a bug.").await?;
170                    error!("DeviantArt deviation missing asset url");
171                }
172            } else {
173                ctx.reply("No Results").await?;
174            }
175        }
176        Err(error) => {
177            ctx.reply(format!("Failed to search \"{query}\": {error:?}"))
178                .await?;
179
180            error!("Failed to search for \"{query}\" on deviantart: {error:?}");
181        }
182    }
183
184    Ok(())
185}
