42
43impl PoiseData {
44    pub async fn new(config: Arc<Config>, database: Database) -> anyhow::Result<Self> {
45        let cache_dir = config.cache_dir();
46
47        let deviantart_client = DeviantartClient::new(&config, database.clone())
48            .await
49            .context("Failed to create deviantart client")?;
50        let encoder_task = EncoderTask::new();
51        let rule34_client = Rule34Client::new(
52            config.rule34.user_id,
53            &config.rule34.api_key,
54            database.clone(),
55        );
56        let sauce_nao_client = SauceNaoClient::new(&config.sauce_nao.api_key);
57        let tic_tac_toe_data = TicTacToeData::new(database.clone());
58        let tiktok_data = TikTokData::new(&cache_dir, encoder_task.clone())
59            .await
60            .context("Failed to init tiktok data")?;
61        let ugg_client = UggClient::new(&cache_dir)
62            .await
63            .context("Failed to init ugg client")?;
64
65        Ok(Self {
66            config,
67            database,
68            deviantart_client,
69            encoder_task,
70            iqdb_client: IqdbClient::new(),
71            nekos_client: NekosClient::new(),
72            quizizz_client: QuizizzClient::new(),
73            r6tracker_client: R6TrackerClient::new(),
74            rule34_client,
75            sauce_nao_client,
76            tic_tac_toe_data,
77            tiktok_data,
78            ugg_client,
79            urban_client: UrbanClient::new(),
80            xkcd_client: xkcd::Client::new(),
81            yodaspeak_client: yodaspeak::Client::new(),
82        })
83    }
84
85    /// Shutdown anything that needs to be shut down.
86    ///
87    /// Errors are logged to the console,
88    /// but not returned to the user as it is assumed that they don't matter in the middle of a shutdown.
89    pub async fn shutdown(&self) {
90        if let Err(error) = self
91            .encoder_task
92            .shutdown()
93            .await
94            .context("Failed to shutdown encoder task")
95        {
96            error!("{error:?}");
97        }
98    }
99}
