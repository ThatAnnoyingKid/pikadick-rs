52    pub async fn upsert_rule34_posts_and_query_stat(
53        &self,
54        posts: Vec<Rule34Post>,
55        query_stat: Rule34QueryStat,
56    ) -> anyhow::Result<()> {
57        if posts.is_empty() {
58            return Ok(());
59        }
60
61        self.database
62            .write(move |database| {
63                let transaction = database.transaction()?;
64
65                let mut post_tag_ids = Vec::new();
66
67                {
68                    let mut query = transaction.prepare_cached(UPSERT_RULE34_TAG)?;
69                    for post in posts.iter() {
70                        let tags: HashSet<_> =
71                            post.tags.split(' ').filter(|tag| !tag.is_empty()).collect();
72                        let mut tag_ids = HashSet::with_capacity(tags.len());
73                        for tag in tags {
74                            let id: u64 = query.query_one(
75                                named_params! {
76                                    ":name": tag,
77                                },
78                                |row| row.get("id"),
79                            )?;
80                            tag_ids.insert(id);
81                        }
82                        post_tag_ids.push(tag_ids);
83                    }
84                }
85
86                {
87                    let mut query = transaction.prepare_cached(UPSERT_RULE34_POST)?;
88                    for post in posts.iter() {
89                        query.execute(named_params! {
90                            ":id": post.id,
91                            ":tags": post.tags,
92                            ":file_url": post.file_url,
93                            ":last_fetched": post.last_fetched.as_millisecond(),
94                        })?;
95                    }
96                }
97
98                {
99                    let mut query = transaction.prepare_cached(CLEAR_RULE34_POST_TAGS)?;
100                    for post in posts.iter() {
101                        query.execute(named_params! {
102                            ":id": post.id,
103                        })?;
104                    }
105                }
106
107                {
108                    let mut query = transaction.prepare_cached(INSERT_RULE34_POST_TAG)?;
109                    for (post, tag_ids) in posts.iter().zip(post_tag_ids.iter()) {
110                        for tag_id in tag_ids.iter() {
111                            query.execute(named_params! {
112                                ":post_id": post.id,
113                                ":tag_id": tag_id,
114                            })?;
115                        }
116                    }
117                }
118
119                {
120                    transaction
121                        .prepare_cached(UPSERT_RULE34_QUERY_STAT)?
122                        .execute(named_params! {
123                            ":query": query_stat.query,
124                            ":queries_since_last_fetch": query_stat.queries_since_last_fetch,
125                            ":last_fetched": JiffTimestampWrapper(query_stat.last_fetched),
126                        })?;
127                }
128
129                transaction.commit()?;
130                anyhow::Ok(())
131            })
132            .await?
133    }

135    /// Get a random rule34 post by tag, along with the query stats.
136    pub async fn get_random_rule34_post_and_query_stat(
137        &self,
138        query: Option<String>,
139        random_seed: i64,
140        limit: Option<usize>,
141    ) -> anyhow::Result<(Vec<Rule34Post>, Option<Rule34QueryStat>)> {
142        self.database
143            .write(move |database| {
144                let transaction = database.transaction()?;
145                let results = transaction
146                    .prepare_cached(GET_RANDOM_RULE34_POST_SQL)?
147                    .query_map(
148                        named_params! {
149                            ":random_seed": random_seed,
150                            ":tag_name": query,
151                        },
152                        Rule34Post::from_row,
153                    )?
154                    .take(limit.unwrap_or(10))
155                    .collect::<Result<Vec<_>, _>>()?;
156
157                let query_stat_key = query.as_deref().unwrap_or("");
158                let query_stat = transaction
159                    .prepare_cached(GET_RULE34_QUERY_STAT)?
160                    .query_one(
161                        named_params! {
162                            ":query": query_stat_key,
163                        },
164                        Rule34QueryStat::from_row,
165                    )
166                    .optional()?;
167
168                if query_stat.is_some() {
169                    transaction
170                        .prepare_cached(ADD_QUERIES_RULE34_QUERY_STAT)?
171                        .execute(named_params! {
172                            ":query": query_stat_key,
173                            ":value": 1,
174                        })?;
175                }
176
177                anyhow::Ok((results, query_stat))
178            })
179            .await?
180    }
