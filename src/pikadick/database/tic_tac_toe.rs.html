112/// The response for making a tic-tac-toe move

127fn get_tic_tac_toe_game(
128    txn: &Transaction<'_>,
129    guild_id: Option<GuildId>,
130    user_id: TicTacToePlayer,
131) -> RusqliteResult<Option<(i64, TicTacToeGame)>> {
132    txn.prepare_cached(GET_TIC_TAC_TOE_GAME_SQL)?
133        .query_row(
134            named_params! {
135                ":guild_id": OptionGuildIdString(guild_id),
136                ":user_id": user_id
137            },
138            |row| {
139                Ok((
140                    row.get(0)?,
141                    TicTacToeGame {
142                        board: Board::decode_u16(row.get(1)?),
143                        x_player: row.get(2)?,
144                        o_player: row.get(3)?,
145                    },
146                ))
147            },
148        )
149        .optional()
150}

176/// Set a tic-tac-toe game as a draw as part of a larger transaction, consuming it.
177fn set_draw_tic_tac_toe_game(
178    txn: Transaction<'_>,
179    id: i64,
180    guild_id: Option<GuildId>,
181    game: TicTacToeGame,
182) -> anyhow::Result<()> {
183    delete_tic_tac_toe_game(&txn, id).context("Failed to delete game")?;
184
185    if let (TicTacToePlayer::User(x_player), TicTacToePlayer::User(o_player)) =
186        (game.x_player, game.o_player)
187    {
188        create_user_score_data(&txn, guild_id, x_player)?;
189        create_user_score_data(&txn, guild_id, o_player)?;
190
191        txn.prepare_cached(INCREMENT_TIES_SCORE_TIC_TAC_TOE_SQL)?
192            .execute(params![
193                OptionGuildIdString(guild_id),
194                i64::from(x_player),
195                i64::from(o_player)
196            ])?;
197    }
198
199    txn.commit().context("Failed to commit")?;
200
201    Ok(())
202}

204/// Set a tic-tac-toe game as a win as part of a larger transaction, consuming it.
205fn set_win_tic_tac_toe_game(
206    txn: Transaction<'_>,
207    id: i64,
208    guild_id: Option<GuildId>,
209    winner: TicTacToePlayer,
210    loser: TicTacToePlayer,
211) -> anyhow::Result<()> {
212    delete_tic_tac_toe_game(&txn, id).context("Failed to delete game")?;
213
214    if let (TicTacToePlayer::User(winner), TicTacToePlayer::User(loser)) = (winner, loser) {
215        create_user_score_data(&txn, guild_id, winner)?;
216        create_user_score_data(&txn, guild_id, loser)?;
217
218        txn.prepare_cached(INCREMENT_WINS_SCORE_TIC_TAC_TOE_SQL)?
219            .execute(params![OptionGuildIdString(guild_id), i64::from(winner)])?;
220        txn.prepare_cached(INCREMENT_LOSSES_SCORE_TIC_TAC_TOE_SQL)?
221            .execute(params![OptionGuildIdString(guild_id), i64::from(loser)])?;
222    }
223
224    txn.commit().context("Failed to commit")?;
225
226    Ok(())
227}

230    /// Create a new tic-tac-toe game.
231    pub async fn create_tic_tac_toe_game(
232        &self,
233        guild_id: Option<GuildId>,
234        author: TicTacToePlayer,
235        author_team: tic_tac_toe::Team,
236        opponent: TicTacToePlayer,
237    ) -> Result<TicTacToeGame, TicTacToeCreateGameError> {
238        let (x_player, o_player) = if author_team == tic_tac_toe::Team::X {
239            (author, opponent)
240        } else {
241            (opponent, author)
242        };
243
244        self.write(move |db| {
245            let transaction = db
246                .transaction_with_behavior(TransactionBehavior::Immediate)
247                .context("Failed to create transaction")
248                .map_err(TicTacToeCreateGameError::Database)?;
249
250            let check_in_game_result: Option<(TicTacToePlayer, TicTacToePlayer)> = transaction
251                .prepare_cached(CHECK_IN_TIC_TAC_TOE_GAME_SQL)
252                .context("Failed to prepare query")
253                .map_err(TicTacToeCreateGameError::Database)?
254                .query_row(
255                    named_params! {
256                        ":guild_id": OptionGuildIdString(guild_id),
257                        ":author": author,
258                        ":opponent": opponent,
259                    },
260                    |row| Ok((row.get(0)?, row.get(1)?)),
261                )
262                .optional()
263                .context("Failed to query if in game")
264                .map_err(TicTacToeCreateGameError::Database)?;
265
266            if let Some((maybe_x_player_in_game, maybe_o_player_in_game)) = check_in_game_result {
267                if maybe_x_player_in_game == author || maybe_o_player_in_game == author {
268                    return Err(TicTacToeCreateGameError::AuthorInGame);
269                }
270
271                if maybe_x_player_in_game == opponent || maybe_o_player_in_game == opponent {
272                    return Err(TicTacToeCreateGameError::OpponentInGame);
273                }
274            }
275
276            let mut game = TicTacToeGame::new(x_player, o_player);
277
278            // TODO: Iteratively perform AI steps?
279            if x_player.is_computer() {
280                let (_score, index) = tic_tac_toe::minimax(game.board, tic_tac_toe::NUM_TILES);
281                game.board = game.board.set(index, Some(tic_tac_toe::Team::X));
282            }
283
284            let board = game.board.encode_u16();
285            transaction
286                .prepare_cached(CREATE_TIC_TAC_TOE_GAME_SQL)
287                .context("Failed to prepare query")
288                .map_err(TicTacToeCreateGameError::Database)?
289                .execute(params![
290                    board,
291                    x_player,
292                    o_player,
293                    OptionGuildIdString(guild_id)
294                ])
295                .context("Failed to create game in database")
296                .map_err(TicTacToeCreateGameError::Database)?;
297
298            transaction
299                .commit()
300                .context("Failed to commit")
301                .map_err(TicTacToeCreateGameError::Database)?;
302
303            Ok(game)
304        })
305        .await
306        .context("Database access failed")
307        .map_err(TicTacToeCreateGameError::Database)?
308    }

310    /// Try to make a tic-tac-toe move
311    pub async fn try_tic_tac_toe_move(
312        &self,
313        guild_id: Option<GuildId>,
314        player: TicTacToePlayer,
315        move_index: u8,
316    ) -> Result<TicTacToeTryMoveResponse, TicTacToeTryMoveError> {
317        self.write(move |db| {
318            let transaction = db
319                .transaction_with_behavior(TransactionBehavior::Immediate)
320                .context("Failed to create transaction")
321                .map_err(TicTacToeTryMoveError::Database)?;
322
323            let (id, mut game) = get_tic_tac_toe_game(&transaction, guild_id, player)
324                .context("Failed to get game")
325                .map_err(TicTacToeTryMoveError::Database)?
326                .ok_or(TicTacToeTryMoveError::NotInAGame)?;
327
328            let player_turn = game.get_player_turn();
329            if player != player_turn {
330                return Err(TicTacToeTryMoveError::InvalidTurn);
331            }
332
333            let team_turn = game.get_team_turn();
334            let move_successful = game.try_move(move_index, team_turn);
335            if !move_successful {
336                return Err(TicTacToeTryMoveError::InvalidMove);
337            }
338
339            if let Some(winner_team) = game.board.get_winner() {
340                let winner = game.get_player(winner_team);
341                let loser = game.get_player(winner_team.inverse());
342
343                set_win_tic_tac_toe_game(transaction, id, guild_id, winner, loser)
344                    .map_err(TicTacToeTryMoveError::Database)?;
345
346                return Ok(TicTacToeTryMoveResponse::Winner {
347                    game,
348                    winner,
349                    loser,
350                });
351            }
352
353            if game.board.is_draw() {
354                set_draw_tic_tac_toe_game(transaction, id, guild_id, game)
355                    .map_err(TicTacToeTryMoveError::Database)?;
356                return Ok(TicTacToeTryMoveResponse::Tie { game });
357            }
358
359            let opponent = game.get_player_turn();
360            if opponent == TicTacToePlayer::Computer {
361                let (_score, index) = tic_tac_toe::minimax(game.board, tic_tac_toe::NUM_TILES);
362                game.board = game.board.set(index, Some(team_turn.inverse()));
363
364                if let Some(winner_team) = game.board.get_winner() {
365                    let winner = game.get_player(winner_team);
366                    let loser = game.get_player(winner_team.inverse());
367
368                    set_win_tic_tac_toe_game(transaction, id, guild_id, winner, loser)
369                        .map_err(TicTacToeTryMoveError::Database)?;
370
371                    return Ok(TicTacToeTryMoveResponse::Winner {
372                        game,
373                        winner,
374                        loser,
375                    });
376                }
377
378                if game.board.is_draw() {
379                    set_draw_tic_tac_toe_game(transaction, id, guild_id, game)
380                        .map_err(TicTacToeTryMoveError::Database)?;
381                    return Ok(TicTacToeTryMoveResponse::Tie { game });
382                }
383            }
384
385            update_tic_tac_toe_game(&transaction, id, game.board)
386                .context("Failed to update game")
387                .map_err(TicTacToeTryMoveError::Database)?;
388
389            transaction
390                .commit()
391                .context("Failed to commit")
392                .map_err(TicTacToeTryMoveError::Database)?;
393
394            Ok(TicTacToeTryMoveResponse::NextTurn { game })
395        })
396        .await
397        .context("Database access failed to join")
398        .map_err(TicTacToeTryMoveError::Database)?
399    }

415    /// Try to concede a Tic-Tac-Toe game.
416    ///
417    /// # Returns
418    /// Returns the game if it existed
419    pub async fn concede_tic_tac_toe_game(
420        &self,
421        guild_id: Option<GuildId>,
422        player: UserId,
423    ) -> anyhow::Result<Option<TicTacToeGame>> {
424        self.write(move |db| {
425            let txn = db.transaction()?;
426            let ret =
427                get_tic_tac_toe_game(&txn, guild_id, player.into()).context("Failed to query")?;
428
429            if let Some((id, game)) = ret {
430                delete_tic_tac_toe_game(&txn, id).context("Failed to delete game")?;
431
432                let conceding_player = TicTacToePlayer::from(player);
433                let opponent = game
434                    .get_opponent(conceding_player)
435                    .context("Missing opponent")?;
436
437                if let (TicTacToePlayer::User(conceding_player), TicTacToePlayer::User(opponent)) =
438                    (conceding_player, opponent)
439                {
440                    txn.prepare_cached(INCREMENT_CONCEDES_SCORE_TIC_TAC_TOE_SQL)?
441                        .execute(params![
442                            OptionGuildIdString(guild_id),
443                            i64::from(conceding_player)
444                        ])?;
445
446                    txn.prepare_cached(INCREMENT_WINS_SCORE_TIC_TAC_TOE_SQL)?
447                        .execute(params![OptionGuildIdString(guild_id), i64::from(opponent)])?;
448                }
449            }
450
451            txn.commit()
452                .context("Failed to commit")
453                .map(|_| ret.map(|ret| ret.1))
454        })
455        .await?
456    }

458    /// Get the user's Tic-Tac-Toe scores
459    pub async fn get_tic_tac_toe_score(
460        &self,
461        guild_id: Option<GuildId>,
462        player: UserId,
463    ) -> anyhow::Result<TicTacToeScore> {
464        self.read(move |db| {
465            let txn = db.transaction()?;
466            create_user_score_data(&txn, guild_id, player)?;
467            let ret = txn.prepare_cached(GET_TIC_TAC_TOE_SCORE_SQL)?.query_row(
468                params![OptionGuildIdString(guild_id), i64::from(player)],
469                |row| {
470                    Ok(TicTacToeScore {
471                        wins: row.get(0)?,
472                        losses: row.get(1)?,
473                        ties: row.get(2)?,
474                        concedes: row.get(3)?,
475                    })
476                },
477            )?;
478            txn.commit().context("Failed to commit").map(|_| ret)
479        })
480        .await?
481    }
